   ```typescript
   private applyTheme(): void {
     const host = this as HTMLElement;
     if (this._theme.primaryColor) {
       host.style.setProperty('--fs-primary-color', this._theme.primaryColor);
     }
     if (this._theme.background) {
       host.style.setProperty('--fs-background', this._theme.background);
     }
     if (this._theme.font) {
       host.style.setProperty('--fs-font', this._theme.font);
     }

     // tear down any previous media listener
     if (this._colorSchemeMedia && this._colorSchemeListener) {
       this._colorSchemeMedia.removeEventListener('change', this._colorSchemeListener);
       this._colorSchemeMedia = null;
       this._colorSchemeListener = null;
     }

     const scheme = this._theme.colorScheme ?? 'auto';

     if (scheme === 'dark') {
       host.setAttribute('data-color-scheme', 'dark');
     } else if (scheme === 'light') {
       host.removeAttribute('data-color-scheme');
     } else {
       // auto: follow prefers-color-scheme
       const applyAuto = () => {
         if (window.matchMedia('(prefers-color-scheme: dark)').matches) {
           host.setAttribute('data-color-scheme', 'dark');
         } else {
           host.removeAttribute('data-color-scheme');
         }
       };
       applyAuto();
       this._colorSchemeMedia = window.matchMedia('(prefers-color-scheme: dark)');
       this._colorSchemeListener = applyAuto;
       this._colorSchemeMedia.addEventListener('change', this._colorSchemeListener);
     }
   }
   ```

```typescript
// MUI theme-bridge helper for @nirioppai/feature-suggestions
// Usage: import { applyMuiTheme } from '@nirioppai/feature-suggestions/themes/mui'
// Peer dep: @mui/material (types only — never imported at runtime)

/** Minimal slice of MUI Theme needed by this helper (avoids hard MUI dep). */
export interface MuiThemeSlice {
  palette: {
    mode: 'light' | 'dark';
    primary: { main: string };
    background: { default: string; paper: string };
    text: { primary: string; secondary: string };
    error: { main: string; light: string };
    divider: string;
  };
  typography?: {
    fontFamily?: string;
    fontSize?: number;
  };
}

/**
 * Apply a MUI theme's palette to a feature-suggestions element as --fs-* vars.
 * @returns unsubscribe — call to stop re-applying (e.g. before re-mounting).
 */
export function applyMuiTheme(
  el: HTMLElement,
  theme: MuiThemeSlice,
): () => void {
  const set = (name: string, value: string) =>
    el.style.setProperty(name, value);

  const apply = (t: MuiThemeSlice) => {
    const isDark = t.palette.mode === 'dark';
    set('--fs-primary-color', t.palette.primary.main);
    set('--fs-background', t.palette.background.default);
    set('--fs-surface', t.palette.background.paper);
    set('--fs-surface-elevated', t.palette.background.paper);
    set(
      '--fs-surface-muted',
      isDark ? 'rgba(255,255,255,0.08)' : 'rgba(0,0,0,0.06)',
    );
    set('--fs-text-color', t.palette.text.primary);
    set('--fs-text-muted', t.palette.text.secondary);
    set('--fs-text-strong', t.palette.text.primary);
    set('--fs-border', t.palette.divider);
    set('--fs-border-strong', t.palette.divider);
    set('--fs-border-focus', t.palette.primary.main);
    set('--fs-error-color', t.palette.error.main);
    set('--fs-error-bg', t.palette.error.light + '26'); // ~15% alpha
    set('--fs-error-border', t.palette.error.light);
    set('--fs-backdrop', isDark ? 'rgba(0,0,0,0.7)' : 'rgba(0,0,0,0.45)');
    set('--fs-modal-bg', t.palette.background.paper);
    if (t.typography?.fontFamily) set('--fs-font', t.typography.fontFamily);
    // colorScheme drives the built-in dark preset for everything else
    el.setAttribute('data-color-scheme', isDark ? 'dark' : 'light');
  };

  apply(theme);

  // Return no-op unsubscribe (MUI theme changes require re-calling applyMuiTheme)
  return () => {};
}
```

```typescript
// Tailwind theme-bridge helper for @nirioppai/feature-suggestions
// Usage: import { applyTailwindTheme } from '@nirioppai/feature-suggestions/themes/tailwind'

export interface TailwindThemeOptions {
  colorScheme?: 'light' | 'dark' | 'auto';
  primary?: string;
  fontFamily?: string;
  radiusScale?: 'sm' | 'md' | 'lg';
  statusPalette?: 'tailwind-default' | 'pastel' | 'high-contrast';

    '--fs-status-planned-bg': '#1e40af',

    '--fs-status-in-progress-fg': '#fef3c7',
    '--fs-status-completed-bg': '#166534',
    '--fs-status-completed-fg': '#dcfce7',
    '--fs-status-declined-bg': '#991b1b',
    '--fs-status-declined-fg': '#fee2e2',
  },
} as const;

const RADIUS_SCALES = {
  sm: {
    '--fs-radius-sm': '2px',
    '--fs-radius-md': '4px',
    '--fs-radius-lg': '6px',
  },
  md: {
    '--fs-radius-sm': '6px',
    '--fs-radius-md': '8px',
    '--fs-radius-lg': '12px',
  },
  lg: {
    '--fs-radius-sm': '8px',
    '--fs-radius-md': '12px',
    '--fs-radius-lg': '16px',
  },
} as const;

/**
 * Apply a Tailwind-flavoured theme to a feature-suggestions element.
 * @returns unsubscribe — removes any dark-mode mutation observer.
 */
export function applyTailwindTheme(
  el: HTMLElement,
  opts: TailwindThemeOptions = {},
): () => void {
  const set = (name: string, value: string) =>
    el.style.setProperty(name, value);

  const {
    primary = 'rgb(99 102 241)',
    fontFamily = 'inherit',
    radiusScale = 'md',
    statusPalette = 'tailwind-default',
    colorScheme = 'auto',
  } = opts;

  set('--fs-primary-color', primary);
  set('--fs-font', fontFamily);

  const radii = RADIUS_SCALES[radiusScale];
  for (const [k, v] of Object.entries(radii)) set(k, v);

  const palette = STATUS_PALETTES[statusPalette];
  for (const [k, v] of Object.entries(palette)) set(k, v);

  const resolveScheme = (): 'light' | 'dark' => {
    if (colorScheme === 'dark') return 'dark';
    if (colorScheme === 'light') return 'light';
    return document.documentElement.classList.contains('dark')
      ? 'dark'
      : 'light';
  };

  const applyScheme = () => {
    el.setAttribute('data-color-scheme', resolveScheme());
  };
  applyScheme();

  if (colorScheme === 'auto') {
    const observer = new MutationObserver(applyScheme);
    observer.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ['class'],
    });
    return () => observer.disconnect();
  }

  return () => {};
}
```
